\chapter{Reciprocal Compactification and the Prime-Tail Certificate}
\label{ch:prime-tail-certificate}

The Hermite ladder of \Cref{ch:hermite-ladder} replaced one localized
test by a countable dense family, but its arithmetic evaluation still used a
finite prime-power cutoff.  This chapter controls that omission for every
fixed finite section.  The key transformation is reciprocal, but it acts on
the logarithmic tail coordinate rather than on a zeta zero.

\section{The omitted arithmetic tail}

For Hermite orders $m,n$, define the omitted prime-power contribution
\[
R_{mn}(Q)=-\frac1{2\pi}\sum_{q>Q}\frac{\Lambda(q)}{\sqrt q}
\left[
\widehat H_{mn}\!\left(\frac{\log q}{2\pi}\right)+
\widehat H_{mn}\!\left(-\frac{\log q}{2\pi}\right)
\right].
\]
The Hermite linearization coefficients and degrees are
\[
 c_{mnk}=2^k k!\binom mk\binom nk,
 \qquad
 d_{mnk}=m+n-2k.
\]
With the normalization of \Cref{ch:hermite-ladder}, the absolute
Fourier prefactor simplifies to
\[
 A_{mn}=\frac1{\sqrt{2^{m+n}m!n!}}.
\]
At $x=\log q/(2\pi)$, the Gaussian-polynomial transform therefore satisfies
\[
\left|\widehat H_{mn}(\pm x)\right|
\le A_{mn}e^{-\kappa_q^2/4}
\sum_{k=0}^{\min(m,n)}c_{mnk}\kappa_q^{d_{mnk}},
\qquad
\kappa_q=\frac{\log q}{w}.
\]

\section{The monotone integral majorant}

For a non-negative integer $d$, put
\[
 g_d(x)=x^{-1/2}(\log x)^{d+1}
 \exp\!\left[-\frac{(\log x)^2}{4w^2}\right].
\]

\begin{lemma}[Monotone tail threshold]
\label{lem:tail-monotone-threshold}
The function $g_d$ is decreasing whenever
\[
 \log x\ge\tau_d(w)
 :=\frac{\sqrt{w^4+8w^2(d+1)}-w^2}{2}.
\]
\end{lemma}

\begin{proof}
Writing $u=\log x$ gives
\[
 \frac{d}{du}\log g_d(e^u)
 =\frac{d+1}{u}-\frac12-\frac{u}{2w^2}.
\]
The right-hand side is non-positive precisely when
$u^2+w^2u-2w^2(d+1)\ge0$.  Its positive root is $\tau_d(w)$.
\end{proof}

Since $\Lambda(q)\le\log q$, and prime powers form a subset of the positive
integers, the decreasing-tail integral test yields the following exact
majorant.

\begin{theorem}[Fixed-entry prime-tail certificate]
\label{thm:prime-tail-entry-certificate}
Assume $\log Q\ge\tau_{m+n}(w)$.  Then
\[
 |R_{mn}(Q)|\le B_{mn}(Q,w),
\]
where
\[
 B_{mn}(Q,w)=\frac{A_{mn}}{\pi}
 \sum_{k=0}^{\min(m,n)}c_{mnk}J_{d_{mnk}}(Q,w)
\]
and
\[
 J_d(Q,w)=\frac1{w^d}\int_Q^\infty
 (\log x)^{d+1}x^{-1/2}
 \exp\!\left[-\frac{(\log x)^2}{4w^2}\right]dx.
\]
\end{theorem}

\begin{proof}
Apply the Fourier majorant to both signs of the transform.  The factor two
cancels the $2$ in the explicit-formula coefficient $1/(2\pi)$.  Replace
$\Lambda(q)$ by $\log q$, enlarge the prime-power sum to all integers, and
apply \Cref{lem:tail-monotone-threshold} term by term.
\end{proof}

\section{The reciprocal map}

Set
\[
 u=\log x,
 \qquad
 z_{\rm t}=\frac1u.
\]
The suffix distinguishes this tail coordinate from the centred zeta coordinate
and from the odds coordinate of \Cref{ch:zero-undefined}.  The map sends
\[
 [\log Q,\infty)
 \longrightarrow
 \left(0,\frac1{\log Q}\right].
\]

\begin{theorem}[Reciprocal compactification]
\label{thm:tail-reciprocal-compactification}
For every non-negative integer $d$,
\[
J_d(Q,w)=\frac1{w^d}
\int_0^{1/\log Q}
 z_{\rm t}^{-(d+3)}
\exp\!\left[-\frac1{4w^2z_{\rm t}^2}+\frac1{2z_{\rm t}}\right]
 dz_{\rm t}.
\]
The integrand extends to a smooth function at $z_{\rm t}=0$ by assigning the
value zero, and every derivative at that endpoint vanishes.
\end{theorem}

\begin{proof}
The integral identity follows from $z_{\rm t}=1/u$ and
$du=-z_{\rm t}^{-2}dz_{\rm t}$.  Near zero, the exponent is bounded above by
$-c/z_{\rm t}^2$ for some $c>0$.  Every derivative is a finite sum of a
rational power of $z_{\rm t}$ times the same exponential, so it tends to zero
faster than any power.
\end{proof}

Thus the infinite arithmetic tail becomes a compact interval with a flat
endpoint.  The map $1/z$ is being used as a computational compactification;
it does not relocate or assume anything about zeta zeros.

\section{Closed incomplete-gamma form}

Completing the square,
\[
 -\frac{u^2}{4w^2}+\frac u2
 =-\frac{(u-w^2)^2}{4w^2}+\frac{w^2}{4},
\]
and expanding the shifted integer power gives an exact finite expression.

\begin{theorem}[Incomplete-gamma tail identity]
\label{thm:tail-gamma-identity}
If $\log Q>w^2$, then
\[
\begin{aligned}
J_d(Q,w)
={}&\frac{e^{w^2/4}}{w^d}
\sum_{j=0}^{d+1}\binom{d+1}{j}
 w^{2(d+1-j)}2^jw^{j+1}\\
&\times\Gamma\!\left(
 \frac{j+1}{2},
 \left(\frac{\log Q-w^2}{2w}\right)^2
\right).
\end{aligned}
\]
\end{theorem}

\begin{proof}
Put $v=u-w^2$, expand $(v+w^2)^{d+1}$, and use
\[
 \int_a^\infty v^j e^{-v^2/(4w^2)}\,dv
 =2^jw^{j+1}\Gamma\!\left(\frac{j+1}{2},\frac{a^2}{4w^2}\right).
\]
\end{proof}

The executable receipt evaluates the log-half-line integral, the reciprocal
compact integral, and this gamma sum independently.

\section{Finite-section norm enclosure}

Let $E_N(Q)$ be the omitted $N\times N$ prime-tail matrix, and let
$B_N(Q)$ contain the entrywise bounds from
\Cref{thm:prime-tail-entry-certificate}.

\begin{corollary}[Operator-norm and eigenvalue enclosure]
\label{cor:tail-operator-envelope}
For every fixed finite section,
\[
 \|E_N(Q)\|_2
 \le \max_{0\le m<N}\sum_{n=0}^{N-1}B_{mn}(Q,w)
 =:\varepsilon_N(Q,w).
\]
Consequently,
\[
 \lambda_{\min}(W_N^{(\infty)})
 \ge
 \lambda_{\min}(W_N^{(Q)})-\varepsilon_N(Q,w).
\]
\end{corollary}

\begin{proof}
The entrywise majorant and symmetry give
$\|E_N\|_1,\|E_N\|_\infty\le\max_m\sum_nB_{mn}$.  Apply
$\|E_N\|_2\le\sqrt{\|E_N\|_1\|E_N\|_\infty}$ and then Weyl's eigenvalue
perturbation inequality.
\end{proof}

\section{Deterministic receipt}

For the declared profile $w=0.8$, $Q=100000$, the receipt is
\[
\begin{array}{c|cc}
N & \max_{m,n<N}B_{mn} & \varepsilon_N\\
\hline
1 & 4.4511\times10^{-21} & 4.4511\times10^{-21}\\
2 & 4.7479\times10^{-19} & 5.2054\times10^{-19}\\
3 & 2.5805\times10^{-17} & 2.9621\times10^{-17}\\
4 & 9.5233\times10^{-16} & 1.1312\times10^{-15}\\
5 & 2.6835\times10^{-14} & 3.2804\times10^{-14}\\
6 & 6.1560\times10^{-13} & 7.7172\times10^{-13}
\end{array}
\]
All degrees through ten satisfy the monotonicity condition with margin greater
than eight in the log coordinate.  The three integral representations agree
to substantially better than the declared $10^{-30}$ tolerance.  Finite shell
checks over $100000<q\le200000$ also lie below their full-tail majorants.

\section{Claim boundary}

\begin{itemize}
\item \textbf{SOH-L018 (exact):} the logarithmic prime tail admits reciprocal
compactification to a finite interval with a flat endpoint.
\item \textbf{SOH-L019 (exact):} each Hermite entry has the stated
incomplete-gamma majorant under the explicit monotonicity condition.
\item \textbf{SOH-L020 (exact finite-section theorem):} entrywise certificates
give an operator-norm and Weyl eigenvalue enclosure for every fixed section.
\item \textbf{SOH-N005 (numerical):} the declared $N\le6$, $Q=100000$ envelope
has maximum norm bound $7.7172\times10^{-13}$.
\end{itemize}

The result closes controlled cutoff removal for each declared fixed finite
section.  It does not prove positivity of those sections, does not provide a
useful bound uniform as $N\to\infty$, and does not prove that a null mode forces
native theta-shell closure.  Those are the remaining parts of
\texttt{SOH-C005}.
