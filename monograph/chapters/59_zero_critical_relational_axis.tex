\chapter{The Zero Axis: Three Complementary Minimal Proof Paths}
\label{chap:zero-critical-relational-axis}

\section{Statement}

Let \(\rho\) be an arbitrary non-trivial zero of the completed Riemann zeta
function.  Under A0 from \Cref{ax:rz-zero}, this chapter proves
\[
\boxed{\Re\rho=\frac12.}
\]
Since \(\rho\) is arbitrary,
\[
\boxed{\mathrm{A0}\Longrightarrow\mathrm{RH}.}
\]

The proof is deliberately presented through three complementary coordinate
systems so that no single representation carries the whole argument by
definition.

\section{Path I: relational fixed point}

Write
\[
\sigma=\Re s,
\qquad
x=\sigma-\frac12.
\]
For a normalized relation/complement pair, exchange is
\[
J(\sigma)=1-\sigma.
\]
By \Cref{thm:rz-fixed-point},
\[
J(\sigma)=\sigma
\iff
\sigma=\frac12
\iff
x=0.
\]

Thus the zero axis of a normalized complementary relation is forced to the
unique exchange-invariant barycenter.

Applied to the completed-zeta horizontal symmetry
\[
s\longmapsto1-\bar s,
\]
the fixed axis is
\[
\boxed{\Re s=\frac12.}
\]

Symmetry by itself allows an off-axis pair.  A0 supplies the extra semantic
content required here: a genuine zero of the admitted observable is a zero of
its canonical relational defect, not merely one member of a symmetric pair.

\section{Path II: Shannon/KL action cost}

The information defect is
\[
D_H(\sigma)
=
\ln2-H_2(\sigma)
=
D_{\rm KL}\!\left(
(\sigma,1-\sigma)
\middle\|
(1/2,1/2)
\right).
\]
Hence
\[
D_H(\sigma)=0
\iff
\sigma=\frac12.
\]
Furthermore
\[
D_H''(\sigma)=\frac1{\sigma(1-\sigma)}\ge4,
\]
and therefore
\[
\boxed{
D_H(\sigma)
\ge
2\left(\sigma-\frac12\right)^2.
}
\]

Let the canonical relational energy contain this defect together with only
non-negative dynamical/holonomy contributions:
\[
E_{\rm rel}(s)
=
D_H(\Re s)
+
\mathcal E_{\rm WT}(s)
+
E_{\rm other,+}(s),
\qquad
\mathcal E_{\rm WT},E_{\rm other,+}\ge0.
\]
Then
\[
E_{\rm rel}(s)\ge D_H(\Re s)\ge2(\Re s-1/2)^2.
\]

By A0, if \(\xi(\rho)=0\), then
\[
E_{\rm rel}(\rho)=0.
\]
The preceding inequality forces
\[
\boxed{\Re\rho=\frac12.}
\]

This path needs no projective coordinate.

\section{Path III: reciprocal--conjugation and \(U(1)\)}

Define
\[
\Omega(s)=\frac{s}{1-s},
\qquad
\Delta_{\rm RC}(u)
=
\left|u^{-1}-\bar u\right|^2.
\]

The formal analytic layer proves, for \(s\ne0,1\),
\[
\boxed{
\Delta_{\rm RC}(\Omega(s))
=
\frac{4(\Re s-1/2)^2}
{|s|^2|1-s|^2}.
}
\tag{59.1}
\]
Consequently
\[
\boxed{
\Delta_{\rm RC}(\Omega(s))=0
\iff
\Re s=\frac12.
}
\tag{59.2}
\]

Combining \Cref{thm:rz-shannon} with (59.1) gives the exact pointwise
coercive estimate
\[
\boxed{
D_H(\Re s)
\ge
\frac{|s|^2|1-s|^2}{2}\,
\Delta_{\rm RC}(\Omega(s)).
}
\tag{59.3}
\]
Hence the full relational energy obeys
\[
\boxed{
E_{\rm rel}(s)
\ge
\frac{|s|^2|1-s|^2}{2}\,
\Delta_{\rm RC}(\Omega(s)).
}
\tag{59.4}
\]

For a non-trivial zero \(\rho\), one has \(\rho\ne0,1\), so
\[
c(\rho):=
\frac{|\rho|^2|1-\rho|^2}{2}
>0.
\]
A0 gives \(E_{\rm rel}(\rho)=0\).  Therefore
\[
0
\le
c(\rho)\Delta_{\rm RC}(\Omega(\rho))
\le
E_{\rm rel}(\rho)
=
0.
\]
Thus
\[
\Delta_{\rm RC}(\Omega(\rho))=0,
\]
and (59.2) yields
\[
\boxed{\Re\rho=\frac12.}
\]

\subsection{\(U(1)\) half-turn cross-check}

The normalized Berry closure satisfies
\[
q_B=-(1-\sigma)\pmod1.
\]
For the canonical representative \(0<\sigma<1\),
\[
q_B=\sigma,
\]
so
\[
q_B-\frac12
=
\sigma-\frac12
=
x.
\]
At the zero axis,
\[
q_B=\frac12,
\qquad
U_B=e^{2\pi iq_B}=-1.
\]
Thus the projective transverse displacement and the normalized \(U(1)\)
half-turn displacement are the same scalar defect in different coordinates.

The White-Thread potential
\[
V_{\rm WT}
=
\sum_e
K_e^{(1)}(1-\cos\Delta_e)
+
K_e^{(1/2)}
\left(1-\cos\frac{\Delta_e}{2}\right)
\]
is non-negative, as is the associated mechanical energy
\[
\mathcal E_{\rm WT}
=
\sum_i\frac{I_i}{2}\dot\Theta_i^2
+
V_{\rm WT}.
\]
Therefore the \(U(1)\) sector strengthens the coercive bound; it is not an
additional axiom.

\section{The three-way closure}

The three paths meet at the same locus:
\[
\boxed{
x=0
\iff
\sigma=\frac12
\iff
D_H(\sigma)=0
\iff
\Delta_{\rm RC}(\Omega(s))=0.
}
\]

Equivalently,
\[
\boxed{
0_{\rm relational}
\equiv
\frac12_{\rm complement}
\equiv
\ln2_{\rm Shannon\ maximum}
\equiv
\text{projective self-dual axis}.
}
\]

The paths are complementary:

\begin{enumerate}
  \item Path I derives the zero axis from relation/complement exchange.
  \item Path II derives the same axis as the unique zero of a strongly convex
  information/action defect.
  \item Path III derives it as the zero set of an exact projective defect and
  identifies the same displacement in the normalized \(U(1)\) half-turn
  coordinate.
\end{enumerate}

\section{Axiomatic Occam RH theorem}

\begin{theorem}[SOH-RZ006: pointwise-coercive zero-energy endgame]
If \(E:\C\to\R\) and \(c:\C\to\R\) satisfy, at every non-trivial zeta zero
\(\rho\),
\[
c(\rho)>0,
\qquad
E(\rho)=0,
\qquad
c(\rho)\Delta_{\rm RC}(\Omega(\rho))\le E(\rho),
\]
then the Riemann Hypothesis follows.
\end{theorem}

\begin{proof}
At every non-trivial zero,
\[
0
\le
c(\rho)\Delta_{\rm RC}(\Omega(\rho))
\le
0.
\]
Since \(c(\rho)>0\), the defect vanishes.  Equation (59.2) gives
\(\Re\rho=1/2\).  As \(\rho\) is arbitrary, RH follows.
\end{proof}

This theorem is formalized in
\texttt{SecretOfAHalfFormal/RadialDefect.lean} in both uniform- and
pointwise-coercive forms.

\begin{theorem}[The Zero Axis theorem]
\label{thm:zero-axis-rh}
Assume A0.  Then every non-trivial zero of the completed Riemann zeta function
lies on the critical line:
\[
\boxed{
\mathrm{A0}\Longrightarrow\mathrm{RH}.
}
\]
\end{theorem}

\begin{proof}
Let \(\rho\) be any non-trivial zero.  A0 gives
\(E_{\rm rel}(\rho)=0\).  Equation (59.4) supplies the pointwise coercive
inequality with
\[
c(\rho)=\frac{|\rho|^2|1-\rho|^2}{2}>0.
\]
SOH-RZ006 therefore gives
\[
\Re\rho=\frac12.
\]
Since \(\rho\) was arbitrary, RH follows.
\end{proof}

zero is forced to be the vanishing point of relation.  Normalized
relation/complement exchange fixes that point at \(1/2\); Shannon/KL geometry

reciprocal--conjugation defect identifies precisely the same axis.
