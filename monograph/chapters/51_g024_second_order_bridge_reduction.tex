\section{Normalized bridge measure}
Recall
\[
C(u)=\int_{\mathbb R}r^2K(u+r)K(u-r)\,dr,
\qquad L=-\log K.
\]
For $u>0$ define the probability measure
\[
\boxed{d\mu_u(r)=\frac{r^2K(u+r)K(u-r)}{C(u)}\,dr.}
\]
Define
\[
\boxed{A_u(r)=L'(u+r)+L'(u-r)},
\qquad
\boxed{B_u(r)=L''(u+r)+L''(u-r)}.
\]

\section{Logarithmic correlation derivatives}
Differentiation under the integral gives
\[
C'(u)=-\int_{\mathbb R}A_u(r)r^2K(u+r)K(u-r)\,dr.
\]
With
\[
R(u)=-\frac{C'(u)}{C(u)},
\]
one has
\[
\boxed{R(u)=\mathbb E_{\mu_u}[A_u].}
\]
For a differentiable bridge observable $F_u$,
\[
\frac{d}{du}\mathbb E_{\mu_u}[F_u]
=\mathbb E_{\mu_u}[\partial_uF_u]-\operatorname{Cov}_{\mu_u}(F_u,A_u).
\]
Taking $F_u=A_u$ gives the exact curvature--variance identity
\[
\boxed{R'(u)=\mathbb E_{\mu_u}[B_u]-\operatorname{Var}_{\mu_u}(A_u).}
\]

\section{External tilt surplus}
Put
\[
a=|y|,\qquad T_y(u)=2a\tanh(2au),
\qquad\boxed{N_y(u)=R(u)-T_y(u)}.
\]
Since $D_y(u)=\cosh(2yu)C(u)$,
\[
\boxed{-\frac{D_y'(u)}{D_y(u)}=N_y(u)},
\qquad
\boxed{T_y'(u)=4a^2\operatorname{sech}^2(2au)}.
\]
Chapter~50 proves $N_y>19u$; Chapter~53 later sharpens this to $N_y>33u$.

\section{Exact second-order normal form}
Because $q=u^2$ and $H_y(q)=D_y(u)$,
\[
H_y'(q)=\frac{D_y'(u)}{2u},
\qquad
\boxed{H_y''(q)=\frac{uD_y''(u)-D_y'(u)}{4u^3}.}
\]
Hence
\[
\boxed{H_y''(q)\ge0\iff uD_y''(u)-D_y'(u)\ge0.}
\]
Using $D_y'/D_y=-N_y$ and the bridge identity for $R'$ yields
\[
\boxed{
\frac{4u^3H_y''(u^2)}{H_y(u^2)}
=N_y+u\left[N_y^2+\operatorname{Var}_{\mu_u}(A_u)-\mathbb E_{\mu_u}[B_u]+4a^2\operatorname{sech}^2(2au)\right].
}
\]
This is the exact curvature--variance obstruction used by the later proofs.

\section{Riccati equivalence}
Define
\[
S_y(q)=-\frac{d}{dq}\log H_y(q).
\]
Since $S_y(u^2)=N_y(u)/(2u)$,
\[
\boxed{\frac{H_y''}{H_y}=S_y^2-S_y'},
\qquad
\boxed{H_y''\ge0\iff S_y'\le S_y^2.}
\]
The latter is the compact Riccati form eventually closed in Chapter~54.

\section{Sign anatomy}
The bridge margin consists of the positive first-order surplus $N_y$, the square $N_y^2$, the non-negative variance, the competing mean-curvature term $-\mathbb E[B_u]$, and the non-negative tilt-curvature term. Strong log-concavity alone does not close the inequality because it supplies a lower bound on $B_u$ rather than the required sufficiently sharp upper control. Chapters~52 and 53 add concentration and quantitative curvature estimates; Chapter~54 introduces the fourth-log-curvature certificate that preserves the needed cancellation.

\section{Gaussian identity regression}
For the exact fixture
\[
K(t)=e^{-t^2/2},\qquad L(t)=\frac{t^2}{2},
\]
one has
\[
A_u=2u,\qquad B_u=2,\qquad\operatorname{Var}(A_u)=0.
\]
The implementation compares the boxed bridge identity with direct differentiation of
\[
H_y(q)=\frac{\sqrt\pi}{2}e^{-q}\cosh(2y\sqrt q).
\]
Agreement validates the implemented identity only; it is not a numerical proof for the Riemann kernel.
