\section{Purpose and literature position}
The preceding PF$_3$ generations isolate coefficient-level obstructions. This chapter returns to an RH-equivalent correlation-kernel criterion and develops a sufficient route based on complete monotonicity. The external correlation construction follows Dimitrov and Xu \citep{dimitrovxu2016}; the positive-definite-kernel setting is related to the programme studied by Csordas \citep{csordas2013positive}; and the translation-density step uses Wiener's $L^1$ Tauberian theorem \citep{wiener1932tauberian}.

\section{Full-line Fourier normalization}
The repository kernel satisfies
\[
\xi\!\left(\frac12+z\right)=\int_0^\infty\Phi(t)\cosh(zt)\,dt.
\]
Define
\[
\boxed{K(t)=\frac12\Phi(|t|).}
\]
Then
\[
\boxed{\Xi(x):=\xi\!\left(\frac12+ix\right)=\int_{\mathbb R}K(t)e^{-ixt}\,dt.}
\]

\section{Dimitrov--Xu order-two correlation}
For a suitable even positive kernel define
\[
\nu_2(t)=\int_{\mathbb R}(t-2s)^2K(t-s)K(s)\,ds,
\qquad
\Psi_y(t)=\cosh(ty)\nu_2(t).
\]
For the Riemann kernel, Dimitrov and Xu prove the translation-density characterization
\[
\boxed{RH\iff\mathcal T(\Psi_y)\text{ is dense in }L^1(\mathbb R)\quad\forall\ 0<|y|<\frac12.}
\]
By Wiener's theorem this is equivalent to absence of real zeros of the corresponding Fourier transform.

Set $t=2u$ and $s=u-r$. Then
\[
\boxed{\nu_2(2u)=4C(u)},
\qquad
\boxed{C(u)=\int_{\mathbb R}r^2K(u+r)K(u-r)\,dr}.
\]
Define the external tilt
\[
\boxed{D_y(u)=\cosh(2yu)C(u)}.
\]
Hence $\Psi_y(2u)=4D_y(u)$ and therefore
\[
\boxed{RH\iff\mathcal T(D_y)\text{ is dense in }L^1(\mathbb R)\quad\forall\ 0<|y|<\frac12.}
\]
Since $D_y$ is real, even, integrable, and $\widehat D_y(0)>0$, the target is equivalently
\[
\boxed{\widehat D_y(x)>0\quad\forall x\in\mathbb R,\quad0<|y|<\frac12.}
\]
This is an exact reformulation, not a proof of the inequality.

\section{External Wronskian normal form}
Let
\[
f(z)=\int_{\mathbb R}K(t)e^{-izt}\,dt.
\]
Direct expansion of $(a-b)^2$ in the double Fourier integral gives
\[
\boxed{\widehat\nu_2(z)=2\bigl(f'(z)^2-f(z)f''(z)\bigr).}
\]
Multiplication by $\cosh(ty)$ averages the transforms at $x\pm iy$, so
\[
\boxed{4\widehat D_y(2x)=\Re\!\left(f'(x+iy)^2-f(x+iy)f''(x+iy)\right).}
\]
Thus the direct external target is
\[
\boxed{\Re\!\left(f'(x+iy)^2-f(x+iy)f''(x+iy)\right)>0}
\]
for every real $x$ and every $0<|y|<1/2$.

\section{Complete-monotonicity sufficient route}
Define
\[
\boxed{H_y(q)=D_y(\sqrt q)=\cosh(2y\sqrt q)C(\sqrt q),\qquad q\ge0.}
\]
If
\[
(-1)^mH_y^{(m)}(q)\ge0
\quad\forall m\ge0,\ q\ge0,
\]
then the Hausdorff--Bernstein--Widder representation gives a positive measure $\mu_y$ \citep{schilling2010bernstein} such that
\[
H_y(q)=\int_0^\infty e^{-\lambda q}\,d\mu_y(\lambda).
\]
Consequently
\[
D_y(u)=\int_0^\infty e^{-\lambda u^2}\,d\mu_y(\lambda),
\]
and integrability excludes a non-zero constant component. Fourier transformation gives a strictly positive Gaussian mixture. Therefore
\[
\boxed{\text{complete monotonicity of every }H_y\Longrightarrow RH.}
\]
The implication is exact; full complete monotonicity remains open.

\section{External and internal tilts must not be conflated}
Define instead
\[
\boxed{J_y(u)=\int_{\mathbb R}r^2\cosh(2yr)K(u+r)K(u-r)\,dr.}
\]
The internal Jensen/Laguerre identity is
\[
\boxed{4\widehat J_y(2x)=|f'(x+iy)|^2-\Re\!\left(f''(x+iy)\overline{f(x+iy)}\right).}
\]
The two families are distinct:
\[
\boxed{D_y(u)=\cosh(2yu)C(u)\neq J_y(u)}
\]
for general $y\ne0$, while
\[
\boxed{D_0=J_0=C.}
\]
This distinction is a proof firewall.

\section{First-order complete monotonicity}
SOH-G004 already supplies a quantitative strong-convexity margin. Writing $L=-\log K$, the conservative estimate at this stage is
\[
L''>10.
\]
Differentiation of the centred correlation yields
\[
C'(u)=-\int r^2K(u+r)K(u-r)\bigl[L'(u+r)+L'(u-r)\bigr]dr.
\]
Strong monotonicity of $L'$ gives
\[
-\frac{C'(u)}{C(u)}>20u.
\]
With $u=\sqrt q$,
\[
-\frac{H_y'(q)}{H_y(q)}
>10-\frac{|y|}{u}\tanh(2|y|u)
>\frac{19}{2}.
\]
Therefore:

\begin{theorem}[SOH-G024 first-order complete-monotonicity theorem]
For the actual Riemann kernel and every $0<|y|<1/2$,
\[
\boxed{H_y'(q)<0\qquad(q\ge0).}
\]
The later sharpened estimate in Chapter~53 improves the logarithmic-slope floor from $19/2$ to $33/2$.
\end{theorem}

Integrating the conservative floor gives the already valid Gaussian domination
\[
H_y(q)<H_y(0)e^{-19q/2}.
\]

\section{Second-order normal form and later closure}
Define
\[
S_y(q)=-\frac{d}{dq}\log H_y(q).
\]
Direct differentiation gives
\[
\boxed{\frac{H_y''}{H_y}=S_y^2-S_y'},
\qquad
\boxed{H_y''\ge0\iff S_y'(q)\le S_y(q)^2.}
\]
This Riccati-type inequality is the next exact target at the end of the present chapter. Chapters~51--53 progressively reduce the remaining domain, and Chapter~54 proves $H_y''>0$ globally with the explicit SOH-G024-Q interval dependency.
