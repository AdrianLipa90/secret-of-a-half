\chapter{Third-Order Bridge Cumulant Reduction}
\label{chap:g024-third-order-cumulant}

\status{Exact SOH-G024-S third-order reduction. The first and second signed derivative inequalities are global. SOH-G024-T, proved later and presented in Chapter~56, establishes that the all-order complete-monotonicity hierarchy must fail at some order $m\ge3$. The identities here remain exact, but closing the third-order sign is no longer a sufficient route to RH.}

\section{Signed derivative recurrence}
For
\[
S_y(q)=-\frac{d}{dq}\log H_y(q),
\qquad
F_m(q)=(-1)^m\frac{H_y^{(m)}(q)}{H_y(q)},
\]
direct differentiation gives
\[
\boxed{F_{m+1}=S_yF_m-F_m'.}
\]
Hence
\[
F_1=S_y,
\qquad F_2=S_y^2-S_y',
\qquad
\boxed{F_3=S_y^3-3S_yS_y'+S_y''.}
\]
Chapters~50--54 establish $F_1>0$ and $F_2>0$ globally. Chapter~56 proves that the full hierarchy cannot remain non-negative for all orders.

\section{Radial third derivative}
Put $q=u^2$ and $H_y(q)=D_y(u)$. Since $d/dq=(2u)^{-1}d/du$,
\[
H_y'''(u^2)
=\frac{u^2D_y'''(u)-3uD_y''(u)+3D_y'(u)}{8u^5}.
\]
Thus
\[
\boxed{-H_y'''(u^2)\ge0
\iff -u^2D_y'''+3uD_y''-3D_y'\ge0.}
\]

Let
\[
N_y(u)=-\frac{D_y'(u)}{D_y(u)}.
\]
Then
\[
\frac{D_y''}{D_y}=N_y^2-N_y',
\qquad
\frac{D_y'''}{D_y}=-N_y^3+3N_yN_y'-N_y'',
\]
so
\[
\boxed{
8u^5\frac{-H_y'''(u^2)}{H_y(u^2)}
=u^2N_y^3-3u^2N_yN_y'+u^2N_y''
+3uN_y^2-3uN_y'+3N_y.}
\]

\section{Bridge cumulant identity}
Under
\[
d\mu_u(r)=\frac{r^2K(u+r)K(u-r)}{C(u)}\,dr,
\]
define
\[
A_u=L'(u+r)+L'(u-r),\quad
B_u=L''(u+r)+L''(u-r),
\]
\[
C_u^{(3)}=L'''(u+r)+L'''(u-r).
\]
For differentiable $G_u$,
\[
\frac{d}{du}\mathbb E[G_u]
=\mathbb E[\partial_uG_u]-\operatorname{Cov}(G_u,A_u).
\]
With $R=\mathbb E[A_u]$ and $R'=\mathbb E[B_u]-\operatorname{Var}(A_u)$, exact differentiation yields
\[
\boxed{
R''=\mathbb E[C_u^{(3)}]
-3\operatorname{Cov}(A_u,B_u)
+\mu_3(A_u).}
\]

\section{Equivalent growth form}
Define
\[
M_2(u)=N_y+u(N_y^2-N_y')
=4u^3\frac{H_y''(u^2)}{H_y(u^2)}>0.
\]
Then
\[
M_2'=N_y^2+u(2N_yN_y'-N_y'')
\]
and the third-order margin becomes
\[
\boxed{
8u^5\frac{-H_y'''(u^2)}{H_y(u^2)}
=(uN_y+3)M_2-uM_2'.}
\]
Therefore the local third-order sign is equivalent to
\[
\boxed{\frac{uM_2'}{M_2}\le uN_y+3.}
\]

\section{Status after SOH-G024-T}
The formulas above remain exact and may be useful for locating the first failure of the signed hierarchy. However, Chapter~56 proves that for each fixed $0<|y|<1/2$ there exists some $m\ge3$ and some $q\ge0$ with
\[
(-1)^mH_y^{(m)}(q)<0.
\]
Accordingly, establishing $F_3\ge0$ globally, if true, would merely move the first possible failure to a higher order; it would not restore the complete-monotonicity route.
