\chapter{Compactified Inverse-Boundary Geometry and the Unique Half-Layer}
\label{ch:compactified-inverse-boundary}

\status{Exact global geometric reduction; the zerohood-to-self-duality implication remains open.}

\section{Purpose}

This chapter replaces an unnecessarily local proof emphasis with the global projective geometry already present in the inverse-boundary coordinate.  No Hermite finite-section assumption is used here.

Define
\[
\Omega(s)=\frac{s}{1-s}.
\]
On the Riemann sphere,
\[
\Omega(0)=0,
\qquad
\Omega(1)=\infty.
\]
Thus the two real strip boundaries become the reciprocal projective pair $0\leftrightarrow\infty$.

For the critical-axis reflection
\[
K(s)=1-\overline{s},
\]
we have
\[
\Omega(K(s))=\frac{1}{\overline{\Omega(s)}}.
\]
Therefore, for
\[
R(s):=|\Omega(s)|,
\]
the induced radial involution is exactly
\[
R\longmapsto R^{-1}.
\]

\section{Compactifying zero and infinity}

Define the compactified radial coordinate
\[
\boxed{
q(s):=\frac{R(s)}{1+R(s)}
}
\]
with the extended values $q=0$ at $R=0$ and $q=1$ at $R=\infty$.  Hence
\[
[0,\infty]\longrightarrow[0,1],
\qquad
0\mapsto0,
\qquad
\infty\mapsto1.
\]

Under reciprocal inversion,
\[
q\left(R^{-1}\right)
=
\frac{R^{-1}}{1+R^{-1}}
=
\frac{1}{1+R}
=
1-q(R).
\]
Thus the projective inversion becomes the ordinary complement involution
\[
\boxed{q\mapsto1-q}.
\]

\begin{theorem}[Compactified inverse-boundary half-axis theorem]
The Möbius coordinate $\Omega(s)=s/(1-s)$ followed by the radial compactification $q=|\Omega|/(1+|\Omega|)$ sends the reciprocal boundary pair $0\leftrightarrow\infty$ to $0\leftrightarrow1$, conjugates radial inversion to $q\mapsto1-q$, and sends the critical line $\Re s=1/2$ exactly to the unique self-dual layer $q=1/2$.
\end{theorem}

\begin{proof}
The complement involution $q\mapsto1-q$ has the unique fixed point $q=1/2$.  Moreover,
\[
q=\frac12
\iff
\frac{R}{1+R}=\frac12
\iff
R=1.
\]
By the exact inverse-boundary relation already established in \cref{ch:li-negative-inverse},
\[
R=|\Omega(s)|=1
\iff
\Re s=\frac12.
\]
Therefore the unique compactified self-dual radial layer is precisely the critical line.
\end{proof}

\section{Centred signed radius}

Define
\[
\boxed{
\eta(s):=2q(s)-1=\frac{R(s)-1}{R(s)+1}
}.
\]
Then
\[
R=0\mapsto\eta=-1,
\qquad
R=1\mapsto\eta=0,
\qquad
R=\infty\mapsto\eta=+1.
\]
Reciprocal inversion gives
\[
\boxed{\eta\mapsto-\eta}.
\]

If
\[
B(s)=\log R(s)=\log|\Omega(s)|,
\]
then
\[
\eta(s)
=
\frac{e^{B(s)}-1}{e^{B(s)}+1}
=
\boxed{\tanh\!\left(\frac{B(s)}{2}\right)}.
\]
Consequently,
\[
\boxed{
B=0
\iff
\eta=0
\iff
q=\frac12
\iff
\Re s=\frac12.
}
\]
These are not distinct proof problems; they are equivalent coordinates for the same self-dual geometry.

\section{Completed zeta and the minimal remaining bridge}

Set
\[
X(u):=\xi\!\left(\frac{u}{1+u}\right),
\qquad u=\Omega(s).
\]
The Riemann Hypothesis is exactly equivalent to
\[
X(u)=0\Longrightarrow |u|=1.
\]
Using the compactified coordinate, this becomes
\[
\boxed{
X(u)=0\Longrightarrow q(u)=\frac12
}
\]
or equivalently
\[
\boxed{
X(u)=0\Longrightarrow \eta(u)=0.
}
\]

\begin{problem}[SOH-G001: global self-dual zero closure]
Prove
\[
X(u)=0\Longrightarrow q(u)=\frac12.
\]
Equivalently, prove $X(u)=0\Rightarrow B(u)=0$ or $X(u)=0\Rightarrow\eta(u)=0$.
\end{problem}

\gap{The geometry itself is closed.  The only remaining analytic implication is from transformed zerohood to the unique self-dual radial layer.}

\section{Proof-target firewall}

The following statements are exact and must not be reopened as separate numerical subproblems:
\begin{enumerate}
\item the projective boundary pair is $0\leftrightarrow\infty$;
\item radial inversion is $R\mapsto R^{-1}$;
\item compactification converts inversion to $q\mapsto1-q$;
\item the unique self-dual layer is $q=1/2$;
\item that layer is exactly $\Re s=1/2$;
\item the Li coordinate $z_L=-1/\Omega$ preserves the same unit-radius self-duality.
\end{enumerate}

Finite-Hermite, moving-boundary, and prime-tail constructions may remain in the repository as auxiliary diagnostics, but they are not independent prerequisites for the statement of SOH-G001 and must not displace the global geometric proof target.

\begin{warning}
This chapter is an exact geometric reduction, not by itself a proof of the Riemann Hypothesis.  The implication SOH-G001 remains open until an analytic argument forces every transformed zero onto the unique self-dual layer.
\end{warning}
