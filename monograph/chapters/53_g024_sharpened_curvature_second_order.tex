\section{A stronger channelwise curvature floor}

For the theta channels of SOH-G004,
\[
 -g_n''=h(r_n),
 \qquad
 h(r)=4r+\frac{24r}{(2r-3)^2},
 \qquad
 r_n=\pi n^2e^{2t}>3.
\]
For every $r>3$,
\[
 \boxed{h(r)>19.}
\]
Indeed,
\[
 h(r)-19
 =\frac{16r^3-124r^2+288r-171}{(2r-3)^2}.
\]
Setting $x=r-3\ge0$, the numerator becomes
\[
 16x^3+20x^2-24x+9.
\]
The quadratic $20x^2-24x+9$ has discriminant $-144<0$ and positive leading coefficient, hence is strictly positive; adding $16x^3\ge0$ preserves positivity.

The G004 mixture identity is
\[
 L''=\sum_np_n(-g_n'')-\operatorname{Var}_p(g_n'),
\]
and G004 proves $\operatorname{Var}_p(g_n')<2$. Therefore
\[
 \boxed{L''(t)>17\qquad(t\in\mathbb R).}
\]

\section{Improved first-order and bridge concentration}

The strengthened convexity gives
\[
 -\frac{C'(u)}{C(u)}>34u.
\]
For $a=|y|<1/2$,
\[
 T_y(u)=2a\tanh(2au)<u,
\]
so
\[
 \boxed{N_y(u)>33u.}
\]
Consequently
\[
 \boxed{-\frac{H_y'(q)}{H_y(q)}>\frac{33}{2}.}
\]
This strictly strengthens the earlier $19/2$ floor.

The bridge score hierarchy now gives
\[
 \boxed{
 \mathbb E[r^{2n+2}]
 <\frac{2n+3}{34}\mathbb E[r^{2n}]
 }
\]
and therefore
\[
 \boxed{
 \mathbb E[r^{2n}]<\frac{(2n+1)!!}{34^n},
 \qquad
 \mathbb E[e^{\lambda r^2}]
 \le(1-\lambda/17)^{-3/2}
 }
\]
for $0\le\lambda<17$.

\section{Improved mean bridge-curvature bound}

The companion upper envelope from Chapter~52 remains
\[
 L''(s)<21e^{2|s|}.
\]
Thus
\[
 B_u(r)<42e^{2u+2|r|}.
\]
Using
\[
 2|r|\le3r^2+\frac13
\]
and the improved bridge MGF at $\lambda=3$,
\[
 \mathbb E[B_u]
 <42e^{2u+1/3}\left(\frac{17}{14}\right)^{3/2}.
\]
The elementary exponential-series estimate
\[
 e
 <\sum_{k=0}^{6}\frac1{k!}
 +\frac1{7!}\sum_{j\ge0}8^{-j}
 =\frac{31967}{11760}
 <\frac{87}{32}
\]
gives $e^{1/3}<7/5$. Also
\[
 \left(\frac{17}{14}\right)^{3/2}<\frac{47}{35}.
\]
Hence
\[
 42e^{1/3}\left(\frac{17}{14}\right)^{3/2}
 <42\cdot\frac75\cdot\frac{47}{35}
 =\frac{1974}{25}<79,
\]
so
\[
 \boxed{\mathbb E[B_u]<79e^{2u}.}
\]

\section{Closing the band $1/3\le u\le1/2$}

The exact second-order normal form implies, after discarding only non-negative terms,
\[
 \frac{4u^3H_y''}{H_y}
 >N_y+u\left[N_y^2-\mathbb E[B_u]\right].
\]
Using $N_y>33u$ and $\mathbb E[B_u]<79e^{2u}$,
\[
 \frac{4u^3H_y''}{H_y}
 >u\left[33+1089u^2-79e^{2u}\right].
\]
Define
\[
 F(u)=33+1089u^2-79e^{2u}.
\]
For $1/3\le u\le1/2$,
\[
 F'(u)=2178u-158e^{2u}
 \ge726-158e
 >\frac{4743}{16}>0,
\]
where $e<87/32$ was used. At $u=1/3$,
\[
 F(1/3)=154-79e^{2/3}>0.
\]
Using again $e<87/32$, the endpoint inequality reduces after cubing to
\[
 79^3\,87^2<154^3\,32^2,
\]
that is
\[
 3731812191<3739918336.
\]
Therefore
\[
 \boxed{H_y''(q)>0\qquad(1/9\le q\le1/4).}
\]

Chapter~52 already proves the same strict sign for $q\ge1/4$. Hence:

\begin{theorem}[SOH-G024 sharpened second-order theorem]
For every external Dimitrov--Xu tilt with $0<|y|<1/2$,
\[
 \boxed{H_y''(q)>0\qquad(q\ge1/9).}
\]
\end{theorem}
